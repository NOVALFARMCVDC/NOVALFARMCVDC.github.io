/**
 * Punto de entrada de la API.
 */
function doGet(e) {

  try {

    const accion = String(e.parameter.action || "")
      .trim()
      .toLowerCase();

    switch (accion) {

      case "buscar":
        return buscarParticipantesAPI(e);

      case "participante":
        return obtenerParticipanteAPI(e);

      /* consulta comercial */

      case "resumengestiones":
        return obtenerResumenGestionesAPI(e);

      case "gestiones":
        return listarGestionesAPI(e);

      case "gestion":
        return obtenerGestionAPI(e);

      /* consulta comercial */

      case "estado":
        return respuestaJSON({
          ok: true,
          servicio: "CVDC2026_API",
          estado: "operativo"
        });

      /* codigo para pruebas */
      case "diagnosticob":
        return diagnosticarBaseB();

      case "columnasformdos":
        return diagnosticarColumnasFormDos();
      /* codigo para pruebas */

      default:

        return respuestaJSON({
          ok: false,
          error: "Acción no válida."
        });

    }

  } catch (error) {

    console.error(error);

    return respuestaJSON({
      ok: false,
      error: "Error interno del servicio."
    });

  }

}

/**
 * ============================================================
 * POST - REGISTRO DE GESTIONES COMERCIALES
 * ============================================================
 */
function doPost(e) {

  try {

    if (
      !e ||
      !e.postData ||
      !e.postData.contents
    ) {

      return respuestaJSON({
        ok: false,
        estado: "solicitud_invalida",
        error: "No se recibió información para procesar."
      });

    }


    let datos;

    try {

      datos =
        JSON.parse(
          e.postData.contents
        );

    } catch (error) {

      return respuestaJSON({
        ok: false,
        estado: "json_invalido",
        error: "El cuerpo de la solicitud no contiene JSON válido."
      });

    }


    const accion =
      String(
        datos.action || ""
      )
        .trim()
        .toLowerCase();


    if (
      accion !==
      "registrargestion"
    ) {

      return respuestaJSON({
        ok: false,
        estado: "accion_invalida",
        error: "Acción POST no válida."
      });

    }


    return registrarGestionAPI(
      datos
    );


  } catch (error) {

    console.error(
      "Error doPost:",
      error
    );


    return respuestaJSON({
      ok: false,
      estado: "error_servidor",
      error: "No fue posible procesar la solicitud."
    });

  }

}

/**
 * Registra una gestión comercial.
 */
function registrarGestionAPI(datos) {

  const requestId =
    String(
      datos.requestId || ""
    ).trim();

  const participanteId =
    String(
      datos.participanteId || ""
    ).trim();


  if (!requestId) {

    return respuestaJSON({
      ok: false,
      estado: "request_id_requerido",
      error: "REQUEST_ID es obligatorio."
    });

  }


  if (!participanteId) {

    return respuestaJSON({
      ok: false,
      estado: "participante_requerido",
      error: "El ID del participante es obligatorio."
    });

  }


  /*
   * Antes de escribir en B comprobamos que
   * el participante realmente exista en A.
   */

  const participante =
    buscarParticipanteInterno(
      participanteId
    );


  if (!participante) {

    return respuestaJSON({
      ok: false,
      estado: "participante_no_encontrado",
      error: "El participante no existe en la base principal."
    });

  }


  /*
   * Bloqueo para evitar condiciones de carrera.
   */

  const lock =
    LockService.getScriptLock();


  try {

    lock.waitLock(
      10000
    );


    const gestionExistente =
      buscarGestionPorRequestId(
        requestId
      );


    /*
     * IDEMPOTENCIA
     *
     * Si REQUEST_ID ya existe,
     * NO creamos otra fila.
     */

    if (gestionExistente) {

      return respuestaJSON({

        ok: true,

        estado:
          "gestion_ya_registrada",

        duplicado:
          true,

        idGestion:
          gestionExistente.idGestion,

        requestId:
          requestId,

        idParticipante:
          participanteId

      });

    }


    const idGestion =
      generarIdGestion();

    const gestion =
      normalizarDatosGestion(
        datos
      );


    const fila =
      construirFilaGestion(
        participante,
        gestion,
        idGestion,
        requestId
      );


    const hoja =
      obtenerHojaGestiones();


    hoja
      .getRange(
        hoja.getLastRow() + 1,
        1,
        1,
        fila.length
      )
      .setValues([
        fila
      ]);

    return respuestaJSON({

      ok: true,

      estado:
        "gestion_registrada",

      duplicado:
        false,

      idGestion:
        idGestion,

      requestId:
        requestId,

      idParticipante:
        participanteId

    });


  } catch (error) {

    console.error(
      "Error registrarGestionAPI:",
      error
    );


    return respuestaJSON({

      ok: false,

      estado:
        "error_registro",

      error:
        "No fue posible registrar la gestión comercial."

    });


  } finally {

    try {

      lock.releaseLock();

    } catch (error) {

      // El bloqueo puede no haberse adquirido.

    }

  }

}

/**
 * Busca participantes por:
 *
 * - Nombre
 * - Correo
 * - Teléfono
 *
 * Devuelve máximo CONFIG.MAX_RESULTADOS.
 */
function buscarParticipantesAPI(e) {

  const termino = normalizarTexto(
    e.parameter.q || ""
  );

  if (
    termino.length <
    CONFIG.MIN_CARACTERES_BUSQUEDA
  ) {

    return respuestaJSON({
      ok: false,
      error: "Ingrese al menos 3 caracteres."
    });

  }


  const hoja = obtenerHojaRespuestas();

  const ultimaFila = hoja.getLastRow();

  if (ultimaFila < 2) {

    return respuestaJSON({
      ok: true,
      total: 0,
      resultados: []
    });

  }


  const datos = hoja
    .getRange(
      2,
      1,
      ultimaFila - 1,
      CONFIG.COLUMNAS.POSICION
    )
    .getDisplayValues();


  const resultados = [];


  for (const fila of datos) {

    const participante =
      convertirFilaParticipante(fila);


    // Ignorar filas preparadas pero vacías.
    if (!participanteValido(participante)) {
      continue;
    }


    const nombre =
      normalizarTexto(participante.nombre);

    const correo =
      normalizarTexto(participante.correo);

    const telefono =
      normalizarTexto(participante.telefono);


    const coincide =
      nombre.includes(termino) ||
      correo.includes(termino) ||
      telefono.includes(termino);


    if (!coincide) {
      continue;
    }


    /*
     * Por privacidad, el buscador NO devuelve
     * dirección, género, autorización, etc.
     */
    resultados.push({

      id: participante.id,

      nombre: participante.nombre,

      telefono: participante.telefono,

      correo: participante.correo

    });


    if (
      resultados.length >=
      CONFIG.MAX_RESULTADOS
    ) {
      break;
    }

  }


  return respuestaJSON({

    ok: true,

    total: resultados.length,

    resultados: resultados

  });

}

/**
 * Busca internamente un participante por ID.
 *
 * Devuelve:
 * objeto participante
 * o null.
 */
function buscarParticipanteInterno(id) {

  const idBuscado =
    String(id || "")
      .trim()
      .toLowerCase();


  if (!idBuscado) {

    return null;

  }


  const hoja =
    obtenerHojaRespuestas();

  const ultimaFila =
    hoja.getLastRow();


  if (ultimaFila < 2) {

    return null;

  }


  const ultimaColumna =
    Math.max(
      hoja.getLastColumn(),
      CONFIG.COLUMNAS.POSICION
    );


  const datos =
    hoja
      .getRange(
        2,
        1,
        ultimaFila - 1,
        ultimaColumna
      )
      .getDisplayValues();


  for (
    let i = 0;
    i < datos.length;
    i++
  ) {

    const fila =
      datos[i];

    const participante =
      convertirFilaParticipante(
        fila
      );


    if (
      participanteValido(
        participante
      ) &&
      String(participante.id)
        .trim()
        .toLowerCase() ===
      idBuscado
    ) {

      return participante;

    }

  }


  return null;

}

/**
 * Busca una gestión previamente registrada
 * utilizando REQUEST_ID.
 */
function buscarGestionPorRequestId(
  requestId
) {

  const hoja =
    obtenerHojaGestiones();

  const ultimaFila =
    hoja.getLastRow();


  if (ultimaFila < 2) {

    return null;

  }


  const columnaRequest =
    CONFIG.COLUMNAS_GESTION
      .REQUEST_ID;


  const valores =
    hoja
      .getRange(
        2,
        columnaRequest,
        ultimaFila - 1,
        1
      )
      .getDisplayValues();


  const buscado =
    String(requestId)
      .trim();


  for (
    let i = 0;
    i < valores.length;
    i++
  ) {

    if (
      String(
        valores[i][0]
      ).trim() === buscado
    ) {

      const fila =
        i + 2;


      const idGestion =
        hoja
          .getRange(
            fila,
            CONFIG.COLUMNAS_GESTION
              .ID_GESTION
          )
          .getDisplayValue();


      return {
        fila: fila,
        idGestion: idGestion
      };

    }

  }


  return null;

}

/**
 * Genera el siguiente ID de gestión.
 *
 * Formato:
 * GES-2026-000001
 */
function generarIdGestion() {

  const hoja =
    obtenerHojaGestiones();

  const ultimaFila =
    hoja.getLastRow();


  let maximo =
    0;


  if (ultimaFila >= 2) {

    const valores =
      hoja
        .getRange(
          2,
          CONFIG.COLUMNAS_GESTION
            .ID_GESTION,
          ultimaFila - 1,
          1
        )
        .getDisplayValues();


    valores.forEach(
      function (fila) {

        const valor =
          String(
            fila[0] || ""
          ).trim();


        if (
          !valor.startsWith(
            CONFIG.PREFIJO_GESTION
          )
        ) {

          return;

        }


        const numero =
          parseInt(
            valor.substring(
              CONFIG.PREFIJO_GESTION.length
            ),
            10
          );


        if (
          Number.isFinite(numero) &&
          numero > maximo
        ) {

          maximo =
            numero;

        }

      }
    );

  }


  const siguiente =
    maximo + 1;


  return (
    CONFIG.PREFIJO_GESTION +
    String(siguiente)
      .padStart(
        CONFIG
          .LONGITUD_CONSECUTIVO_GESTION,
        "0"
      )
  );

}

/**
 * Obtiene un participante completo mediante ID_UNICO.
 */
function obtenerParticipanteAPI(e) {

  const idBuscado =
    String(e.parameter.id || "")
      .trim()
      .toUpperCase();


  if (!idBuscado) {

    return respuestaJSON({
      ok: false,
      error: "Debe indicar el ID del participante."
    });

  }


  const hoja = obtenerHojaRespuestas();

  const ultimaFila = hoja.getLastRow();


  if (ultimaFila < 2) {

    return respuestaJSON({
      ok: false,
      error: "No existen participantes registrados."
    });

  }


  const datos = hoja
    .getRange(
      2,
      1,
      ultimaFila - 1,
      CONFIG.COLUMNAS.POSICION
    )
    .getDisplayValues();


  for (const fila of datos) {

    const participante =
      convertirFilaParticipante(fila);


    if (!participanteValido(participante)) {
      continue;
    }


    if (
      String(participante.id)
        .trim()
        .toUpperCase() === idBuscado
    ) {

      return respuestaJSON({

        ok: true,

        participante: participante

      });

    }

  }


  return respuestaJSON({
    ok: false,
    error: "Participante no encontrado."
  });

}

      fila[c.NOMBRE - 1],

    telefono:
      fila[c.TELEFONO - 1],

    correo:
      fila[c.CORREO - 1],

    ciudad:
      fila[c.CIUDAD - 1],

    direccion:
      fila[c.DIRECCION - 1],

    edad:
      fila[c.EDAD - 1],

    genero:
      fila[c.GENERO - 1],

    quienEs:
      fila[c.QUIEN_ES - 1],

    otro:
      fila[c.OTRO - 1],

    empresa:
      fila[c.EMPRESA - 1],

    autorizacion:
      fila[c.AUTORIZACION - 1],

    id:
      fila[c.ID_UNICO - 1]

  };

}

/**
 * Evita considerar como participantes
 * las filas preparadas pero todavía vacías.
 */
function participanteValido(participante) {

  return Boolean(
    participante.timestamp &&
    participante.nombre &&
    participante.id
  );

/**
 * Normaliza texto para permitir búsquedas
 * más tolerantes.
 *
 * Ejemplo:
 *
 * "María Gómez"
 *
 * puede encontrarse escribiendo:
 *
 * "maria"
 */
function normalizarTexto(valor) {

  return String(valor || "")
    .trim()
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "");

}

/**
 * Convierte número de columna de Sheets
 * a su representación alfabética.
 *
 * 1  -> A
 * 26 -> Z
 * 27 -> AA
 * 32 -> AF
 */
function numeroALetraColumna(numero) {

  let resultado =
    "";

  let valor =
    numero;


  while (valor > 0) {

    const resto =
      (valor - 1) % 26;


    resultado =
      String.fromCharCode(
        65 + resto
      ) +
      resultado;


    valor =
      Math.floor(
        (valor - 1) / 26
      );

  }


  return resultado;

}

/**
 * Determina si una fila representa
 * una gestión comercial válida.
 */
function gestionValida(gestion) {

  if (!gestion) {
    return false;
  }


  const estado =
    normalizarTexto(
      gestion.estadoRegistro
    );


  return Boolean(

    gestion.idGestion &&

    gestion.idParticipante &&

    gestion.nombre &&

    gestion.fechaRegistroApi &&

    estado ===
    normalizarTexto(
      CONFIG.ESTADO_REGISTRADO
    )

  );

}

/**
 * ============================================================
 * FASE 4.1
 * RESUMEN EJECUTIVO DE GESTIONES
 * ============================================================
 */
function obtenerResumenGestionesAPI(e) {

  const hoja =
    obtenerHojaGestiones();

  const ultimaFila =
    hoja.getLastRow();


  if (ultimaFila < 2) {

    return respuestaJSON({

      ok: true,

      resumen: {

        totalGestiones: 0,

        interesAlto: 0,

        potencialAlto: 0,

        solicitanVisita: 0,

        entregaMuestra: 0

      }

    });

  }


  const datos =
    hoja
      .getRange(
        2,
        1,
        ultimaFila - 1,
        CONFIG.COLUMNAS_GESTION
          .ESTADO_REGISTRO
      )
      .getDisplayValues();


  let totalGestiones = 0;

  let interesAlto = 0;

  let potencialAlto = 0;

  let solicitanVisita = 0;

  let entregaMuestra = 0;


  datos.forEach(
    function (fila) {

      const gestion =
        convertirFilaGestion(
          fila
        );


      if (
        !gestionValida(
          gestion
        )
      ) {

        return;

      }


      totalGestiones++;


      if (
        normalizarTexto(
          gestion.nivelInteres
        ) === "alto"
      ) {

        interesAlto++;

      }


      if (
        normalizarTexto(
          gestion.potencialCompra
        ) === "alto"
      ) {

        potencialAlto++;

      }


      if (
        normalizarTexto(
          gestion.solicitaVisita
        ) === "si"
      ) {

        solicitanVisita++;

      }


      if (
        normalizarTexto(
          gestion.entregaMuestra
        ) === "si"
      ) {

        entregaMuestra++;

      }

    }
  );


  return respuestaJSON({

    ok: true,

    resumen: {

      totalGestiones:
        totalGestiones,

      interesAlto:
        interesAlto,

      potencialAlto:
        potencialAlto,

      solicitanVisita:
        solicitanVisita,

      entregaMuestra:
        entregaMuestra

    }

  });

}

 * ============================================================
 * FASE 4.1
 * LISTADO CONTROLADO DE GESTIONES
 * ============================================================
 */

function obtenerParametrosGestiones(e) {

  const parametros =
    e && e.parameter
      ? e.parameter
      : {};


  let pagina =
    parseInt(
      parametros.page,
      10
    );


  let limite =
    parseInt(
      parametros.limit,
      10
    );


  if (
    !Number.isInteger(pagina) ||
    pagina < 1
  ) {

    pagina = 1;

  }


  if (
    !Number.isInteger(limite) ||
    limite < 1
  ) {

    limite =
      CONFIG.CONSULTA_GESTIONES
        .LIMITE_PREDETERMINADO;

  }


  limite =
    Math.min(
      limite,
      CONFIG.CONSULTA_GESTIONES
        .LIMITE_MAXIMO
    );


  return {

    pagina:
      pagina,

    limite:
      limite,

    nivel:
      String(
        parametros.nivel || ""
      ).trim(),

    potencial:
      String(
        parametros.potencial || ""
      ).trim(),

    comercial:
      String(
        parametros.comercial || ""
      ).trim(),

    visita:
      String(
        parametros.visita || ""
      ).trim(),

    linea:
      String(
        parametros.linea || ""
      ).trim(),

    buscar:
      String(
        parametros.buscar || ""
      ).trim()

  };

}

function cumpleFiltrosGestion(
  gestion,
  filtros
) {

  if (
    filtros.nivel &&
    normalizarTexto(
      gestion.nivelInteres
    ) !==
    normalizarTexto(
      filtros.nivel
    )
  ) {

    return false;

  }


  if (
    filtros.potencial &&
    normalizarTexto(
      gestion.potencialCompra
    ) !==
    normalizarTexto(
      filtros.potencial
    )
  ) {

    return false;

  }


  if (
    filtros.visita &&
    normalizarTexto(
      gestion.solicitaVisita
    ) !==
    normalizarTexto(
      filtros.visita
    )
  ) {

    return false;

  }


  if (
    filtros.comercial &&
    !normalizarTexto(
      gestion.comercialRegistro
    ).includes(
      normalizarTexto(
        filtros.comercial
      )
    )
  ) {

    return false;

  }


  if (
    filtros.linea
  ) {

    const lineaPrincipal =
      normalizarTexto(
        gestion.lineaInteres
      );

    const otraLinea =
      normalizarTexto(
        gestion.otraLinea
      );

    const buscada =
      normalizarTexto(
        filtros.linea
      );


    if (
      !lineaPrincipal.includes(
        buscada
      ) &&
      !otraLinea.includes(
        buscada
      )
    ) {

      return false;

    }

  }


  if (
    filtros.buscar
  ) {

    const termino =
      normalizarTexto(
        filtros.buscar
      );


    const contenido =
      [

        gestion.idGestion,

        gestion.idParticipante,

        gestion.nombre,

        gestion.empresa,

        gestion.lineaInteres,

        gestion.otraLinea,

        gestion.productosInteres,

        gestion.comercialRegistro,

        gestion.responsableSeguimiento

      ]
        .map(
          normalizarTexto
        )
        .join(" ");


    if (
      !contenido.includes(
        termino
      )
    ) {

      return false;

    }

  }


  return true;

}

function listarGestionesAPI(e) {

    const filtros =
        obtenerParametrosGestiones(
            e
        );


    const hoja =
        obtenerHojaGestiones();


    const ultimaFila =
        hoja.getLastRow();


    if (ultimaFila < 2) {

        return respuestaJSON({

            ok: true,

            total: 0,

            pagina:
                filtros.pagina,

            limite:
                filtros.limite,

            totalPaginas: 0,

            gestiones: []

        });

    }


    const datos =
        hoja
            .getRange(
                2,
                1,
                ultimaFila - 1,
                CONFIG.COLUMNAS_GESTION
                    .ESTADO_REGISTRO
            )
            .getDisplayValues();


    const resultados =
        [];


    for (
        let i = datos.length - 1;
        i >= 0;
        i--
    ) {

        const gestion =
            convertirFilaGestion(
                datos[i]
            );


        if (
            !gestionValida(
                gestion
            )
        ) {

            continue;

        }


        if (
            !cumpleFiltrosGestion(
                gestion,
                filtros
            )
        ) {

            continue;

        }


        resultados.push({

            idGestion:
                gestion.idGestion,

            idParticipante:
                gestion.idParticipante,

            nombre:
                gestion.nombre,

            empresa:
                gestion.empresa,

            lineaInteres:
                gestion.lineaInteres,

            otraLinea:
                gestion.otraLinea,

            nivelInteres:
                gestion.nivelInteres,

            potencialCompra:
                gestion.potencialCompra,

            solicitaVisita:
                gestion.solicitaVisita,

            tiempoCompra:
                gestion.tiempoCompra,

            accionSiguiente:
                gestion.accionSiguiente,

            comercialRegistro:
                gestion.comercialRegistro,

            responsableSeguimiento:
                gestion.responsableSeguimiento,

            fechaRegistroApi:
                gestion.fechaRegistroApi,

            estadoRegistro:
                gestion.estadoRegistro

        });

    }


    const total =
        resultados.length;


    const totalPaginas =
        total === 0
            ? 0
            : Math.ceil(
                total /
                filtros.limite
            );


    const inicio =
        (
            filtros.pagina - 1
        ) *
        filtros.limite;


    const fin =
        inicio +
        filtros.limite;


    const paginaResultados =
        resultados.slice(
            inicio,
            fin
        );


    return respuestaJSON({

        ok: true,

        total:
            total,

        pagina:
            filtros.pagina,

        limite:
            filtros.limite,

        totalPaginas:
            totalPaginas,

        resultadosPagina:
            paginaResultados.length,

        gestiones:
            paginaResultados

    });

}

/**
 * ============================================================
 * FASE 4.1
 * DETALLE INDIVIDUAL DE UNA GESTIÓN
 * ============================================================
 */
function obtenerGestionAPI(e) {

  const idBuscado =
    String(
      e.parameter.id || ""
    )
      .trim()
      .toUpperCase();


  if (!idBuscado) {

    return respuestaJSON({

      ok: false,

      error:
        "Debe indicar el ID de la gestión."

    });

  }


  const hoja =
    obtenerHojaGestiones();

  const ultimaFila =
    hoja.getLastRow();


  if (ultimaFila < 2) {

    return respuestaJSON({

      ok: false,

      error:
        "No existen gestiones registradas."

    });

  }


  const datos =
    hoja
      .getRange(
        2,
        1,
        ultimaFila - 1,
        CONFIG.COLUMNAS_GESTION
          .ESTADO_REGISTRO
      )
      .getDisplayValues();


  for (const fila of datos) {

    const gestion =
      convertirFilaGestion(
        fila
      );


    if (
      String(
        gestion.idGestion
      )
        .trim()
        .toUpperCase() ===
      idBuscado
    ) {

      return respuestaJSON({

        ok: true,

        gestion: gestion

      });

    }

  }


  return respuestaJSON({

    ok: false,

    error:
      "Gestión comercial no encontrada."

  });

}
